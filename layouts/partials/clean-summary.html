{{/* partials/clean-summary.html Returns a summary that always ends on a complete sentence. - Front matter `description` wins if present. - Otherwise: first sentence always; keeps adding sentences (up to 3) while the running total stays within the character limit. Short sentences → you get three. Long ones → you get one. - Never cuts mid-sentence, never adds an ellipsis. - Decimal numbers (5.8, $4.99, 4.5 stars, etc.) are protected from being misread as sentence-ending periods (fixed 2026-09-09 — see "Mine has a 5." bug on the iPhone Duo post, same class of issue as the pre-existing "Dr." abbreviation limitation, which is not fixed by this change). Usage: {{ partial "clean-summary.html" . }} */}} {{- $limit := 180 -}} {{- $maxSentences := 3 -}} {{- $out := "" -}} {{- if .Description -}} {{- $out = .Description -}} {{- else -}} {{- /* htmlUnescape BEFORE plainify — Micro.blog stores curly quotes as entities */ -}} {{- $text := .Content | htmlUnescape | plainify -}} {{- $text = trim (replaceRE `\s+` " " $text) " \n\t​" -}} {{- /* protect digit.digit patterns (5.8-inch, $4.99, 4.5 stars) so the sentence splitter below doesn't mistake that period for a sentence end — restored right before output */ -}} {{- $text = replaceRE `(\d)\.(\d)` "${1}@@DOT@@${2}" $text -}} {{- /* split into sentences: anything up to . ! or ? (optionally followed by a closing quote) */ -}} {{- $sentences := findRE `[^.!?]+[.!?]+["”’)]?` $text -}} {{- if not $sentences -}} {{- /* no punctuation at all (e.g. a one-line micro): use the whole text */ -}} {{- $out = $text -}} {{- else -}} {{- range $i, $s := $sentences -}} {{- $s = trim $s " " -}} {{- if eq $i 0 -}} {{- $out = $s -}} {{- else if and (lt $i $maxSentences) (le (add (len $out) (len $s) 1) $limit) -}} {{- $out = printf "%s %s" $out $s -}} {{- end -}} {{- end -}} {{- end -}} {{- end -}} {{- $out = replaceRE `@@DOT@@` "." $out -}} {{- $out -}}
