$ph) 0 -}} {{- $title := trim $p.Title " \n\t" -}} {{- $cap := partial "clean-summary.html" (dict "page" $p) | plainify | htmlUnescape -}} {{- $cap = trim (replaceRE `\s+` " " $cap) " " -}} {{- $total
